    75→
    76→function formatRupiah(amount: number): string {
    77→  return new Intl.NumberFormat('id-ID', {
    78→    style: 'currency',
    79→    currency: 'IDR',
    80→    minimumFractionDigits: 0,
    81→    maximumFractionDigits: 0,
    82→  }).format(amount);
    83→}
    84→
    85→function formatDate(dateStr: string): string {
    86→  return new Date(dateStr).toLocaleDateString('id-ID', {
    87→    day: '2-digit',
    88→    month: 'short',
    89→    year: 'numeric',
    90→  });
    91→}

   108→// ─── Status Badge ─────────────────────────────────────────────────────────────
   109→
   110→function getStatusBadge(status: string) {
   111→  switch (status) {
   112→    case 'received':
   113→      return (
   114→        <Badge className="bg-emerald-100 text-emerald-700 border-emerald-200 hover:bg-emerald-100">
   115→          Diterima
   116→        </Badge>
   117→      );
   118→    case 'pending':
   119→      return (
   120→        <Badge className="bg-amber-100 text-amber-700 border-amber-200 hover:bg-amber-100">
   121→          Tertunda
   122→        </Badge>
   123→      );
   124→    case 'cancelled':
   125→      return (
   126→        <Badge className="bg-red-100 text-red-700 border-red-200 hover:bg-red-100">
   127→          Dibatalkan
   128→        </Badge>
   129→      );
   130→    default:
   131→      return <Badge variant="secondary">{status}</Badge>;
   132→  }
   133→}

   135→// ─── Form Schema ──────────────────────────────────────────────────────────────
   136→
   137→const purchaseItemSchema = z.object({
   138→  productId: z.string().min(1, 'Produk wajib dipilih'),
   139→  quantity: z.coerce.number().int().min(1, 'Minimal 1'),
   140→  costPrice: z.coerce.number().min(0, 'Harga tidak boleh negatif'),
   141→});
   142→
   143→type PurchaseItemFormData = z.infer<typeof purchaseItemSchema>;
   144→
   145→const purchaseFormSchema = z.object({
   146→  supplierId: z.string().min(1, 'Supplier wajib dipilih'),
   147→  notes: z.string().optional().or(z.literal('')),
   148→  items: z.array(purchaseItemSchema).min(1, 'Minimal 1 item pembelian'),
   149→});
   150→
   151→type PurchaseFormData = z.infer<typeof purchaseFormSchema>;
   152→
   153→// ─── Product Select Field ─────────────────────────────────────────────────────
   154→
   155→interface ProductSelectFieldProps {
   156→  control: Control<PurchaseFormData>;
   157→  index: number;
   158→  products: Product[];
   159→  productsLoading: boolean;
   160→  onProductChange: (index: number, productId: string) => void;
   161→}
   162→
   163→function ProductSelectField({ control, index, products, productsLoading, onProductChange }: ProductSelectFieldProps) {
   164→  return (
   165→    <FormField
   166→      control={control}
   167→      name={`items.${index}.productId`}
   168→      render={({ field }) => (
   169→        <FormItem>
   170→          <FormLabel className="sr-only">Produk</FormLabel>
   171→          <FormControl>
   172→            <Select
   173→              value={field.value}
   174→              onValueChange={(val) => {
   175→                field.onChange(val);
   176→                onProductChange(index, val);
   177→              }}
   178→              disabled={productsLoading}
   179→            >
   180→              <SelectTrigger className="w-full">
   181→                <SelectValue placeholder={productsLoading ? 'Memuat produk...' : 'Pilih produk'} />
   182→              </SelectTrigger>
   183→              <SelectContent>
   184→                {products.map((product) => (
   185→                  <SelectItem key={product.id} value={product.id}>
   186→                    <div className="flex items-center justify-between gap-4 w-full">
   187→                      <span className="truncate">{product.name}</span>
   188→                      <span className="text-xs text-muted-foreground whitespace-nowrap">
   189→                        Stok: {product.stock} {product.unit}
   190→                      </span>
   191→                    </div>
   192→                  </SelectItem>
   193→                ))}
   194→              </SelectContent>
   195→            </Select>
   196→          </FormControl>
   197→          <FormMessage />
   198→        </FormItem>
   199→      )}
   200→    />
   201→  );
   202→}

   205→
   206→export default function PurchasesPage() {
   207→  const queryClient = useQueryClient();
   208→
   209→  // ── Filter State ────────────────────────────────────────────────────────────
   210→  const [search, setSearch] = useState('');
   211→  const [page, setPage] = useState(1);
   212→  const [perPage, setPerPage] = useState(10);
   213→
   214→  // ── Dialog State ────────────────────────────────────────────────────────────
   215→  const [addOpen, setAddOpen] = useState(false);
   216→  const [detailOpen, setDetailOpen] = useState(false);
   217→  const [detailId, setDetailId] = useState<string | null>(null);
   218→
   219→  // ── Form ───────────────────────────────────────────────────────────────────
   220→  const form = useForm<PurchaseFormData>({
   221→    resolver: zodResolver(purchaseFormSchema),
   222→    defaultValues: {
   223→      supplierId: '',
   224→      notes: '',
   225→      items: [{ productId: '', quantity: 1, costPrice: 0 }],
   226→    },
   227→  });
   228→
   229→  const { fields, append, remove } = useFieldArray({
   230→    control: form.control,
   231→    name: 'items',
   232→  });
   233→
   234→  // ── Queries ────────────────────────────────────────────────────────────────
   235→
   236→  const purchasesQuery = useQuery<PaginatedResponse<Purchase>>({
   237→    queryKey: ['purchases', page, perPage, search],
   238→    queryFn: async () => {
   239→      const params = new URLSearchParams({
   240→        page: String(page),
   241→        perPage: String(perPage),
   242→      });
   243→      if (search) params.set('search', search);
   244→      const res = await fetch(`/api/purchases?${params.toString()}`);
   245→      if (!res.ok) throw new Error('Gagal memuat pembelian');
   246→      return res.json();
   247→    },
   248→  });
   249→
   250→  const suppliersQuery = useQuery<Supplier[]>({
   251→    queryKey: ['suppliers-list'],
   252→    queryFn: async () => {
   253→      const res = await fetch('/api/suppliers?perPage=100');
   254→      if (!res.ok) throw new Error('Gagal memuat supplier');
   255→      const data = await res.json();
   256→      return data.data ?? data;
   257→    },
   258→  });
   259→
   260→  const productsQuery = useQuery<Product[]>({
   261→    queryKey: ['products-list'],
   262→    queryFn: async () => {
   263→      const res = await fetch('/api/products?perPage=200');
   264→      if (!res.ok) throw new Error('Gagal memuat produk');
   265→      const data = await res.json();
   266→      return data.data ?? data;
   267→    },
   268→  });
   269→
   270→  const detailQuery = useQuery<Purchase>({
   271→    queryKey: ['purchase-detail', detailId],
   272→    queryFn: async () => {
   273→      if (!detailId) throw new Error('No ID');
   274→      const res = await fetch(`/api/purchases/${detailId}`);
   275→      if (!res.ok) throw new Error('Gagal memuat detail pembelian');
   276→      return res.json();
   277→    },
   278→    enabled: !!detailId,
   279→  });
   280→
   281→  // ── Mutation ────────────────────────────────────────────────────────────────
   282→
   283→  const createMutation = useMutation({
   284→    mutationFn: async (data: PurchaseFormData) => {
   285→      const payload = {
   286→        supplierId: data.supplierId,
   287→        notes: data.notes || undefined,
   288→        items: data.items.map((item) => ({
   289→          productId: item.productId,
   290→          quantity: item.quantity,
   291→          costPrice: item.costPrice,
   292→        })),
   293→      };
   294→      const res = await fetch('/api/purchases', {
   295→        method: 'POST',
   296→        headers: { 'Content-Type': 'application/json' },
   297→        body: JSON.stringify(payload),
   298→      });
   299→      if (!res.ok) {
   300→        const err = await res.json();
   301→        throw new Error(err.error || 'Gagal menambah pembelian');
   302→      }
   303→      return res.json();
   304→    },
   305→    onSuccess: () => {
   306→      toast.success('Pembelian berhasil ditambahkan');
   307→      queryClient.invalidateQueries({ queryKey: ['purchases'] });
   308→      queryClient.invalidateQueries({ queryKey: ['products'] });
   309→      closeAddDialog();
   310→    },
   311→    onError: (error) => {
   312→      toast.error(error.message);
   313→    },
   314→  });
   315→
   316→  // ── Derived Data ────────────────────────────────────────────────────────────
   317→
   318→  const purchases = purchasesQuery.data?.data ?? [];
   319→  const totalPurchases = purchasesQuery.data?.total ?? 0;
   320→  const totalPages = purchasesQuery.data?.totalPages ?? 1;
   321→  const isLoading = purchasesQuery.isLoading || purchasesQuery.isFetching;
   322→  const isMutating = createMutation.isPending;
   323→  const suppliers = suppliersQuery.data ?? [];
   324→  const products = productsQuery.data ?? [];
   325→  const detailData = detailQuery.data;
   326→
   327→  const startItem = (page - 1) * perPage + 1;
   328→  const endItem = Math.min(page * perPage, totalPurchases);
   329→
   330→  // Watch items for grand total
   331→  const watchItems = form.watch('items');
   332→  const grandTotal = useMemo(() => {
   333→    return watchItems.reduce((sum, item) => sum + (item.quantity * item.costPrice), 0);
   334→  }, [watchItems]);
   335→
   336→  // ── Handlers ────────────────────────────────────────────────────────────────
   337→
   338→  const openAddDialog = useCallback(() => {
   339→    form.reset({
   340→      supplierId: '',
   341→      notes: '',
   342→      items: [{ productId: '', quantity: 1, costPrice: 0 }],
   343→    });
   344→    setAddOpen(true);
   345→  }, [form]);
   346→
   347→  const closeAddDialog = useCallback(() => {
   348→    setAddOpen(false);
   349→  }, []);
   350→
   351→  const openDetail = useCallback((id: string) => {
   352→    setDetailId(id);
   353→    setDetailOpen(true);
   354→  }, []);
   355→
   356→  const closeDetail = useCallback(() => {
   357→    setDetailOpen(false);
   358→    setTimeout(() => setDetailId(null), 200);
   359→  }, []);
   360→
   361→  const handleSearchChange = useCallback((value: string) => {
   362→    setSearch(value);
   363→    setPage(1);
   364→  }, []);
   365→
   366→  const handlePerPageChange = useCallback((value: string) => {
   367→    setPerPage(Number(value));
   368→    setPage(1);
   369→  }, []);
   370→
   371→  const handleAddItem = useCallback(() => {
   372→    append({ productId: '', quantity: 1, costPrice: 0 });
   373→  }, [append]);
   374→
   375→  const handleProductChange = useCallback(
   376→    (index: number, productId: string) => {
   377→      const product = products.find((p) => p.id === productId);
   378→      if (product) {
   379→        form.setValue(`items.${index}.costPrice`, product.costPrice, { shouldValidate: true });
   380→      }
   381→    },
   382→    [form, products],
   383→  );
   384→
   385→  const onSubmitForm = useCallback(
   386→    (data: PurchaseFormData) => {
   387→      createMutation.mutate(data);
   388→    },
   389→    [createMutation],
   390→  );
   391→
   392→  // ── Table Columns ─────────────────────────────────────────────────────────
   393→
   394→  const columns = useMemo<ColumnDef<Purchase>[]>(
   395→    () => [
   396→      {
   397→        accessorKey: 'index',
   398→        header: 'No',
   399→        size: 50,
   400→        cell: ({ row }) => (
   401→          <span className="text-muted-foreground text-sm">{row.index + 1}</span>
   402→        ),
   403→      },
   404→      {
   405→        accessorKey: 'invoiceNumber',
   406→        header: 'Invoice',
   407→        size: 170,
   408→        cell: ({ row }) => (
   409→          <span className="font-mono text-sm font-medium">{row.original.invoiceNumber}</span>
   410→        ),
   411→      },
   412→      {
   413→        accessorKey: 'createdAt',
   414→        header: 'Tanggal',
   415→        size: 130,
   416→        cell: ({ row }) => (
   417→          <span className="text-muted-foreground text-sm">{formatDate(row.original.createdAt)}</span>
   418→        ),
   419→      },
   420→      {
   421→        accessorKey: 'supplier',
   422→        header: 'Supplier',
   423→        size: 160,
   424→        cell: ({ row }) => (
   425→          <span className="text-sm">{row.original.supplier?.name || '-'}</span>
   426→        ),
   427→      },
   428→      {
   429→        accessorKey: 'totalAmount',
   430→        header: 'Total',
   431→        size: 150,
   432→        cell: ({ row }) => (
   433→          <span className="font-medium text-sm">{formatRupiah(row.original.totalAmount)}</span>
   434→        ),
   435→      },
   436→      {
   437→        accessorKey: 'status',
   438→        header: 'Status',
   439→        size: 110,
   440→        cell: ({ row }) => getStatusBadge(row.original.status),
   441→      },
   442→      {
   443→        id: 'actions',
   444→        header: 'Aksi',
   445→        size: 80,
   446→        cell: ({ row }) => (
   447→          <Button
   448→            variant="ghost"
   449→            size="sm"
   450→            className="text-emerald-600 hover:text-emerald-700 hover:bg-emerald-50"
   451→            onClick={(e) => {
   452→              e.stopPropagation();
   453→              openDetail(row.original.id);
   454→            }}
   455→          >
   456→            <Eye className="size-4 mr-1" />
   457→            Detail
   458→          </Button>
   459→        ),
   460→      },
   461→    ],
   462→    [openDetail],
   463→  );
   464→
   465→  // ── Table ─────────────────────────────────────────────────────────────────
   466→
   467→  const table = useReactTable({
   468→    data: purchases,
   469→    columns,
   470→    getCoreRowModel: getCoreRowModel(),
   471→    getPaginationRowModel: getPaginationRowModel(),
   472→    manualPagination: true,
   473→    pageCount: totalPages,
   474→    state: {
   475→      pagination: { pageIndex: page - 1, pageSize: perPage },
   476→    },
   477→  });
   478→
   479→  // ── Render ─────────────────────────────────────────────────────────────────
   480→
   481→  return (
   482→    <motion.div
   483→      variants={containerVariants}
   484→      initial="hidden"
   485→      animate="visible"
   486→      className="flex flex-col gap-6 p-4 md:p-6"
   487→    >
   488→      {/* ── Header ─────────────────────────────────────────────────────────── */}
   489→      <motion.div
   490→        variants={itemVariants}
   491→        className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4"
   492→      >
   493→        <div>
   494→          <h1 className="text-2xl font-bold tracking-tight">Pembelian Barang</h1>
   495→          <p className="text-muted-foreground text-sm mt-1">
   496→            Kelola pembelian barang dari supplier
   497→          </p>
   498→        </div>
   499→        <Button onClick={openAddDialog} className="bg-emerald-600 hover:bg-emerald-700">
   500→          <Plus className="size-4" />
   501→          Tambah Pembelian
   502→        </Button>
   503→      </motion.div>
   504→
   505→      {/* ── Search & Table ──────────────────────────────────────────────────── */}
   506→      <motion.div
   507→        variants={itemVariants}
   508→        className="border rounded-xl bg-card shadow-sm"
   509→      >
   510→        <div className="p-4 border-b">
   511→          <div className="relative max-w-sm">
   512→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   513→            <Input
   514→              placeholder="Cari nomor invoice..."
   515→              value={search}
   516→              onChange={(e) => handleSearchChange(e.target.value)}
   517→              className="pl-9"
   518→            />
   519→          </div>
   520→        </div>
   521→
   522→        <div className="overflow-x-auto">
   523→          <Table>
   524→            <TableHeader>
   525→              {table.getHeaderGroups().map((headerGroup) => (
   526→                <TableRow key={headerGroup.id}>
   527→                  {headerGroup.headers.map((header) => (
   528→                    <TableHead key={header.id} style={{ width: header.getSize() }}>
   529→                      {flexRender(header.column.columnDef.header, header.getContext())}
   530→                    </TableHead>
   531→                  ))}
   532→                </TableRow>
   533→              ))}
   534→            </TableHeader>
   535→            <TableBody>
   536→              {isLoading ? (
   537→                Array.from({ length: perPage }).map((_, i) => (
   538→                  <TableRow key={i}>
   539→                    <TableCell><Skeleton className="h-4 w-6" /></TableCell>
   540→                    <TableCell><Skeleton className="h-4 w-[140px]" /></TableCell>
   541→                    <TableCell><Skeleton className="h-4 w-[110px]" /></TableCell>
   542→                    <TableCell><Skeleton className="h-4 w-[130px]" /></TableCell>
   543→                    <TableCell><Skeleton className="h-4 w-[120px]" /></TableCell>
   544→                    <TableCell><Skeleton className="h-5 w-[90px] rounded-full" /></TableCell>
   545→                    <TableCell><Skeleton className="h-8 w-16 rounded-md" /></TableCell>

   552→                      <PackageOpen className="size-10 text-muted-foreground/50" />
   553→                      <p className="text-muted-foreground text-sm">

   560→                </TableRow>

   562→                table.getRowModel().rows.map((row) => (
   563→                  <TableRow
   564→                    key={row.id}
   565→                    className="cursor-pointer hover:bg-muted/50"
   566→                    onClick={() => openDetail(row.original.id)}
   567→                  >
   568→                    {row.getVisibleCells().map((cell) => (
   569→                      <TableCell key={cell.id}>
   570→                        {flexRender(cell.column.columnDef.cell, cell.getContext())}
   571→                      </TableCell>
   572→                    ))}
   573→                  </TableRow>
   574→                ))
   575→              )}
   576→            </TableBody>
   577→          </Table>
   578→        </div>
   579→
   580→        {/* ── Pagination ─────────────────────────────────────────────────── */}
   581→        {totalPurchases > 0 && (
   582→          <div className="flex flex-col sm:flex-row items-center justify-between gap-3 px-4 py-3 border-t">
   583→            <div className="flex items-center gap-3 text-sm text-muted-foreground">
   584→              <span>
   585→                Menampilkan {startItem}-{endItem} dari {totalPurchases}
   586→              </span>
   587→              <Select value={String(perPage)} onValueChange={handlePerPageChange}>
   588→                <SelectTrigger className="w-[70px] h-8">
   589→                  <SelectValue />
   590→                </SelectTrigger>
   591→                <SelectContent>
   592→                  <SelectItem value="10">10</SelectItem>
   593→                  <SelectItem value="25">25</SelectItem>
   594→                  <SelectItem value="50">50</SelectItem>
   595→                </SelectContent>
   596→              </Select>
   597→              <span>per halaman</span>
   598→            </div>
   599→            <div className="flex items-center gap-2">
   600→              <Button
   601→                variant="outline"
   602→                size="sm"
   603→                onClick={() => setPage((p) => Math.max(1, p - 1))}
   604→                disabled={page <= 1}
   605→              >
   606→                <ChevronLeft className="size-4" />
   607→                <span className="sm:hidden">Sebelumnya</span>
   608→              </Button>
   609→              <div className="flex items-center gap-1 text-sm">
