    70→function formatRupiah(amount: number): string {
    71→  return new Intl.NumberFormat('id-ID', {
    72→    style: 'currency',
    73→    currency: 'IDR',
    74→    minimumFractionDigits: 0,
    75→    maximumFractionDigits: 0,
    76→  }).format(amount);
    77→}
    78→
    79→function formatNominal(raw: string): string {
    80→  const digits = raw.replace(/\D/g, '');
    81→  if (!digits) return '';
    82→  return digits.replace(/\B(?=(\d{3})+(?!\d))/g, '.');
    83→}
    84→
    85→function parseNominal(formatted: string): number {
    86→  return parseInt(formatted.replace(/\D/g, ''), 10) || 0;
    87→}
    88→
    89→function formatDateTime(date: string): string {
    90→  return new Date(date).toLocaleDateString('id-ID', {
    91→    day: '2-digit',
    92→    month: '2-digit',
    93→    year: 'numeric',
    94→    hour: '2-digit',
    95→    minute: '2-digit',
    96→  });
    97→}
