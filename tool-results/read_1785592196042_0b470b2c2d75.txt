    68→function formatRupiah(amount: number): string {
    69→  return new Intl.NumberFormat('id-ID', {
    70→    style: 'currency',
    71→    currency: 'IDR',
    72→    minimumFractionDigits: 0,
    73→    maximumFractionDigits: 0,
    74→  }).format(amount);
    75→}
    76→
    77→function formatDate(dateStr: string): string {
    78→  return new Date(dateStr).toLocaleDateString('id-ID', {
    79→    day: '2-digit',
    80→    month: 'short',
    81→    year: 'numeric',
    82→  });
    83→}
    84→
    85→function getDefaultDates() {
    86→  const now = new Date();
    87→  const firstDay = new Date(now.getFullYear(), now.getMonth(), 1);
    88→  return {
    89→    start: firstDay.toISOString().split('T')[0],
    90→    end: now.toISOString().split('T')[0],
    91→  };
    92→}

   213→function PenjualanTab() {
   214→  const defaults = getDefaultDates();
   215→  const [startDate, setStartDate] = useState(defaults.start);
   216→  const [endDate, setEndDate] = useState(defaults.end);
   217→
   218→  const { data, isLoading, isError } = useQuery({
   219→    queryKey: ['report-daily', startDate, endDate],
   220→    queryFn: async () => {
   221→      const params = new URLSearchParams({ type: 'daily' });
   222→      if (startDate) params.set('startDate', startDate);
   223→      if (endDate) params.set('endDate', endDate);
   224→      const res = await fetch(`/api/reports?${params}`);
   225→      if (!res.ok) throw new Error('Gagal memuat laporan');
   226→      const json = await res.json();
   227→      return json.data;
   228→    },
   229→  });
   230→
   231→  // Group transactions by date for chart & table
   232→  const dailyData = useMemo(() => {
   233→    if (!data?.transactions) return [];
   234→    const grouped: Record<string, { date: string; transactions: number; revenue: number }> = {};
   235→    for (const tx of data.transactions) {
   236→      const day = new Date(tx.createdAt).toISOString().split('T')[0];
   237→      if (!grouped[day]) grouped[day] = { date: day, transactions: 0, revenue: 0 };
   238→      grouped[day].transactions += 1;
   239→      grouped[day].revenue += tx.grandTotal;
   240→    }
   241→    return Object.values(grouped).sort((a, b) => a.date.localeCompare(b.date));
   242→  }, [data]);
   243→
   244→  const summary = data?.summary;
   245→  const avgPerTransaction = summary ? (summary.totalSales / (summary.totalTransactions || 1)) : 0;
   246→
   247→  const getExportData = useCallback(() => {
   248→    const columns = ['Tanggal', 'Jumlah Transaksi', 'Total Pendapatan'];
   249→    const rows = dailyData.map((d) => [
   250→      formatDate(d.date),
   251→      d.transactions,
   252→      formatRupiah(d.revenue),
   253→    ]);
   254→    const summaryObj = summary
   255→      ? {
   256→          'Total Transaksi': String(summary.totalTransactions ?? 0),
   257→          'Total Pendapatan': formatRupiah(summary.totalSales ?? 0),
   258→          'Rata-rata / Transaksi': formatRupiah(avgPerTransaction),
   259→        }
   260→      : undefined;
   261→    return { title: 'Laporan Penjualan', subtitle: `${formatDate(startDate)} – ${formatDate(endDate)}`, columns, rows, summary: summaryObj };
   262→  }, [dailyData, summary, avgPerTransaction, startDate, endDate]);
   263→
   264→  const handleExcel = useCallback(() => {
   265→    const { title, columns, rows } = getExportData();
   266→    exportToExcel({ sheetName: title, columns, rows });
   267→  }, [getExportData]);
   268→
   269→  const handlePrintA4 = useCallback(() => {
   270→    const { title, subtitle, columns, rows, summary } = getExportData();
   271→    printReport({ title, subtitle, columns, rows, summary });
   272→  }, [getExportData]);
   273→
   274→  const handlePrintThermal = useCallback(() => {
   275→    const { title, columns, rows, summary } = getExportData();
   276→    printThermal({ storeName: STORE_NAME, title, columns, rows, summary });
   277→  }, [getExportData]);
   278→
   279→  return (
   280→    <div className="space-y-4">
   281→      <Card>
   282→        <CardContent className="p-4">
   283→          <DateRangeFilter
   284→            startDate={startDate}
   285→            endDate={endDate}
   286→            onStartChange={setStartDate}
   287→            onEndChange={setEndDate}
   288→          />
   289→        </CardContent>
   290→      </Card>
   291→
   292→      <ExportButtons onExcel={handleExcel} onPrintA4={handlePrintA4} onPrintThermal={handlePrintThermal} />
   293→
   294→      {/* Summary Cards */}
   295→      <div className="grid gap-4 sm:grid-cols-3">
   296→        <Card>
   297→          <CardContent className="p-4">
   298→            <div className="flex items-center gap-3">
   299→              <div className="rounded-lg bg-primary/10 p-2.5">
   300→                <ShoppingCart className="h-5 w-5 text-primary" />
   301→              </div>
   302→              <div>
   303→                <p className="text-sm text-muted-foreground">Total Transaksi</p>
   304→                {isLoading ? (
   305→                  <Skeleton className="h-6 w-16" />
   306→                ) : (
   307→                  <p className="text-2xl font-bold">
   308→                    {summary?.totalTransactions ?? 0}
   309→                  </p>
   310→                )}
   311→              </div>
   312→            </div>
   313→          </CardContent>
   314→        </Card>
   315→        <Card>
   316→          <CardContent className="p-4">
   317→            <div className="flex items-center gap-3">
   318→              <div className="rounded-lg bg-emerald-500/10 p-2.5">
   319→                <DollarSign className="h-5 w-5 text-emerald-600" />
   320→              </div>

   325→                ) : (
   326→                  <p className="text-2xl font-bold">
   327→                    {formatRupiah(summary?.totalSales ?? 0)}
   328→                  </p>
   329→                )}
   330→              </div>
   331→            </div>
   332→          </CardContent>
   333→        </Card>
   334→        <Card>
   335→          <CardContent className="p-4">
