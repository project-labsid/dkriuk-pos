    70→function formatRupiah(amount: number): string {
    71→  return new Intl.NumberFormat('id-ID', {
    72→    style: 'currency',
    73→    currency: 'IDR',
    74→    minimumFractionDigits: 0,
    75→    maximumFractionDigits: 0,
    76→  }).format(amount);
    77→}

    79→function formatDateTime(date: string): string {
    80→  return new Date(date).toLocaleDateString('id-ID', {
    81→    day: '2-digit',
    82→    month: '2-digit',
    83→    year: 'numeric',
    84→    hour: '2-digit',
    85→    minute: '2-digit',
    86→  });
    87→}
