    83→function formatDate(dateStr: string): string {
    84→  return new Date(dateStr).toLocaleDateString('id-ID', {
    85→    day: '2-digit',
    86→    month: 'short',
    87→    year: 'numeric',
    88→    hour: '2-digit',
    89→    minute: '2-digit',
    90→  });
    91→}
    92→
    93→function formatRupiah(amount: number): string {
    94→  return new Intl.NumberFormat('id-ID', {
    95→    style: 'currency',
    96→    currency: 'IDR',
    97→    minimumFractionDigits: 0,
    98→    maximumFractionDigits: 0,
    99→  }).format(amount);
   100→}

   109→// ─── Product Search Select ────────────────────────────────────────────────────
   110→
   111→function ProductSearchSelect({
   112→  value,
   113→  onChange,
   114→  selectedStock,
   115→}: {
   116→  value: string;
   117→  onChange: (id: string) => void;
   118→  selectedStock?: number;
   119→}) {
   120→  const [open, setOpen] = useState(false);
   121→  const [search, setSearch] = useState('');
   122→
   123→  const { data, isLoading } = useQuery({
   124→    queryKey: ['products-search', search],
   125→    queryFn: async () => {
   126→      const params = new URLSearchParams({ perPage: '20' });
   127→      if (search) params.set('search', search);
   128→      const res = await fetch(`/api/products?${params}`);
   129→      if (!res.ok) throw new Error('Gagal memuat produk');
   130→      const json = (await res.json()) as PaginatedResponse<Product>;
   131→      return json.data;
   132→    },
   133→  });
   134→
   135→  const selected = data?.find((p) => p.id === value);
   136→
   137→  return (
   138→    <div className="space-y-1.5">
   139→      <Label>Pilih Produk</Label>
   140→      <Popover open={open} onOpenChange={setOpen}>
   141→        <PopoverTrigger asChild>
   142→          <Button
   143→            variant="outline"
   144→            role="combobox"
   145→            aria-expanded={open}
   146→            className="w-full justify-between font-normal"
   147→          >
   148→            {selected ? (
   149→              <span className="flex items-center gap-2 truncate">
   150→                <Package className="h-4 w-4 shrink-0 text-muted-foreground" />
   151→                <span className="truncate">{selected.name}</span>
   152→                {selectedStock !== undefined && (
   153→                  <Badge variant="secondary" className="shrink-0 text-xs">
   154→                    Stok: {selectedStock}
   155→                  </Badge>
   156→                )}
   157→              </span>
   158→            ) : (
   159→              <span className="text-muted-foreground">Cari produk...</span>
   160→            )}
   161→            <ChevronRight className="ml-2 h-4 w-4 shrink-0 opacity-50" />
   162→          </Button>
   163→        </PopoverTrigger>
   164→        <PopoverContent className="w-[300px] p-0" align="start">
   165→          <Command>
   166→            <CommandInput
   167→              placeholder="Cari nama produk..."
   168→              value={search}
   169→              onValueChange={setSearch}
   170→            />
   171→            <CommandList>
   172→              <CommandEmpty>
   173→                {isLoading ? (
   174→                  <div className="flex items-center justify-center py-6">
   175→                    <Loader2 className="h-4 w-4 animate-spin" />
   176→                  </div>
   177→                ) : (
   178→                  'Produk tidak ditemukan'
   179→                )}
   180→              </CommandEmpty>
   181→              <CommandGroup>
   182→                {data?.map((product) => (
   183→                  <CommandItem
   184→                    key={product.id}
   185→                  value={`${product.name} ${product.sku || ''} ${product.barcode || ''}`}
   186→                  onSelect={() => {
   187→                    onChange(product.id);
   188→                    setOpen(false);
   189→                  }}
   190→                  className="flex items-center gap-2"
   191→                >
   192→                    <Package className="h-4 w-4 shrink-0 text-muted-foreground" />
   193→                    <div className="flex flex-col overflow-hidden">
   194→                      <span className="truncate text-sm">{product.name}</span>
   195→                      <span className="text-xs text-muted-foreground">
   196→                        Stok: {product.stock} · {formatRupiah(product.sellPrice)}
   197→                      </span>
   198→                    </div>
   199→                  </CommandItem>
   200→                ))}
   201→              </CommandGroup>
   202→            </CommandList>
   203→          </Command>
   204→        </PopoverContent>
   205→      </Popover>
   206→    </div>
   207→  );
   208→}

   212→function RiwayatStokTab() {
   213→  const [page, setPage] = useState(1);
   214→  const [perPage, setPerPage] = useState(20);
   215→  const [typeFilter, setTypeFilter] = useState('');
   216→  const [search, setSearch] = useState('');
   217→  const [startDate, setStartDate] = useState('');
   218→  const [endDate, setEndDate] = useState('');
   219→  const [searchInput, setSearchInput] = useState('');
   220→
   221→  const { data, isLoading, isError } = useQuery({
   222→    queryKey: ['stock-history', page, perPage, typeFilter, search, startDate, endDate],
   223→    queryFn: async () => {
   224→      const params = new URLSearchParams({
   225→        page: String(page),
   226→        perPage: String(perPage),
   227→      });
   228→      if (typeFilter) params.set('type', typeFilter);
   229→      if (search) params.set('search', search);
   230→      if (startDate) params.set('startDate', startDate);
   231→      if (endDate) params.set('endDate', endDate);

   233→      if (!res.ok) throw new Error('Gagal memuat riwayat stok');
   234→      return res.json() as Promise<PaginatedResponse<StockAdjustment & { user?: { id: string; name: string; email: string } }>>;

   239→    setSearch(searchInput);
   240→    setPage(1);

   244→    setTypeFilter('');
   245→    setSearch('');
   246→    setSearchInput('');

   248→    setEndDate('');
   249→    setPage(1);

   264→                  value={searchInput}
   265→                  onChange={(e) => setSearchInput(e.target.value)}
   266→                  onKeyDown={(e) => e.key === 'Enter' && handleSearch()}

   268→                />
   269→              </div>

   272→              <Label className="text-xs text-muted-foreground">Tipe</Label>

   430→const stockInSchema = z.object({
   431→  productId: z.string().min(1, 'Produk wajib dipilih'),
   432→  quantity: z.coerce.number().int().min(1, 'Jumlah minimal 1'),
   433→  reason: z.string().optional(),
   434→});
   435→
   436→type StockInForm = z.infer<typeof stockInSchema>;
   437→
   438→function StokMasukTab() {
   439→  const queryClient = useQueryClient();
   440→  const [selectedProductId, setSelectedProductId] = useState('');
   441→
   442→  const form = useForm<StockInForm>({
   443→    resolver: zodResolver(stockInSchema),
   444→    defaultValues: { productId: '', quantity: 1, reason: '' },
   445→  });
   446→
   447→  const { data: selectedProduct } = useQuery({
   448→    queryKey: ['product', selectedProductId],
   449→    queryFn: async () => {
   450→      if (!selectedProductId) return null;
   451→      const res = await fetch(`/api/products/${selectedProductId}`);
   452→      if (!res.ok) return null;
   453→      const json = await res.json();
   454→      return json.data as Product;
   455→    },
   456→    enabled: !!selectedProductId,
   457→  });
   458→
   459→  const mutation = useMutation({
   460→    mutationFn: async (values: StockInForm) => {
   461→      const res = await fetch('/api/stock', {
   462→        method: 'POST',
   463→        headers: { 'Content-Type': 'application/json' },
   464→        body: JSON.stringify({ ...values, type: 'in', userId: '' }),
   465→      });
   466→      const json = await res.json();
   467→      if (!res.ok) throw new Error(json.error || 'Gagal menambahkan stok');
   468→      return json;
   469→    },
   470→    onSuccess: () => {
   471→      toast.success('Stok masuk berhasil ditambahkan');
   472→      queryClient.invalidateQueries({ queryKey: ['stock-history'] });
   473→      queryClient.invalidateQueries({ queryKey: ['products-search'] });
   474→      queryClient.invalidateQueries({ queryKey: ['product', selectedProductId] });
   475→      form.reset();
   476→      setSelectedProductId('');
   477→    },
   478→    onError: (err) => toast.error(err.message),
   479→  });
   480→
   481→  const onSubmit = (values: StockInForm) => mutation.mutate(values);
   482→
   483→  return (
   484→    <motion.div
   485→      initial={{ opacity: 0, y: 10 }}
   486→      animate={{ opacity: 1, y: 0 }}
   487→      className="mx-auto max-w-lg"
   488→    >
   489→      <Card>
   490→        <CardHeader>
   491→          <CardTitle className="flex items-center gap-2">
   492→            <ArrowDownCircle className="h-5 w-5 text-emerald-500" />
   493→            Tambah Stok Masuk
   494→          </CardTitle>
   495→          <CardDescription>
   496→            Tambahkan jumlah stok baru untuk produk
   497→          </CardDescription>
   498→        </CardHeader>
   499→        <CardContent>
   500→          <Form {...form}>
   501→            <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-4">
   502→              <ProductSearchSelect
   503→                value={selectedProductId}
   504→                onChange={(id) => {
   505→                  setSelectedProductId(id);
   506→                  form.setValue('productId', id);
   507→                }}
   508→              />
   509→              {selectedProduct && (
   510→                <div className="rounded-lg border bg-muted/50 p-3 text-sm">
   511→                  <div className="grid grid-cols-2 gap-2">
   512→                    <span className="text-muted-foreground">Stok Saat Ini</span>
   513→                    <span className="font-mono font-medium">{selectedProduct.stock} {selectedProduct.unit}</span>
   514→                    <span className="text-muted-foreground">Harga Jual</span>
   515→                    <span className="font-mono">{formatRupiah(selectedProduct.sellPrice)}</span>
   516→                  </div>
   517→                </div>
   518→              )}
   519→              <FormField
   520→                control={form.control}
   521→                name="quantity"
   522→                render={({ field }) => (
   523→                  <FormItem>
   524→                    <FormLabel>Jumlah</FormLabel>
   525→                    <FormControl>
   526→                      <Input
   527→                        type="number"
   528→                        min={1}
   529→                        placeholder="Masukkan jumlah"
   530→                        {...field}
   531→                      />
   532→                    </FormControl>
   533→                    <FormMessage />
   534→                  </FormItem>
   535→                )}
   536→              />
   537→              <FormField
   538→                control={form.control}
   539→                name="reason"
   540→                render={({ field }) => (
   541→                  <FormItem>
   542→                    <FormLabel>Alasan (opsional)</FormLabel>
   543→                    <FormControl>
   544→                      <Textarea
   545→                        placeholder="Misal: Pembelian dari supplier"
   546→                        rows={3}
   547→                        {...field}
   548→                      />
   549→                    </FormControl>
   550→                    <FormMessage />
   551→                  </FormItem>
   552→                )}
   553→              />
   554→              <Button
   555→                type="submit"
   556→                className="w-full"
   557→                disabled={mutation.isPending}
   558→              >
   559→                {mutation.isPending ? (
   560→                  <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   561→                ) : (
   562→                  <CheckCircle2 className="mr-2 h-4 w-4" />
   563→                )}
   564→                Simpan Stok Masuk
   565→              </Button>
   566→            </form>
   567→          </Form>
   568→        </CardContent>
   569→      </Card>
   570→    </motion.div>
   571→  );
   572→}

   584→function StokKeluarTab() {

   586→  const [selectedProductId, setSelectedProductId] = useState('');
   587→
   588→  const form = useForm<StockOutForm>({
   589→    resolver: zodResolver(stockOutSchema),
   590→    defaultValues: { productId: '', quantity: 1, reason: '' },
   591→  });
   592→
   593→  const { data: selectedProduct } = useQuery({
   594→    queryKey: ['product', selectedProductId],
   595→    queryFn: async () => {
   596→      if (!selectedProductId) return null;
   597→      const res = await fetch(`/api/products/${selectedProductId}`);
   598→      if (!res.ok) return null;
   599→      const json = await res.json();
   600→      return json.data as Product;
   601→    },
   602→    enabled: !!selectedProductId,
   603→  });
   604→
   605→  const mutation = useMutation({
   606→    mutationFn: async (values: StockOutForm) => {
   607→      const res = await fetch('/api/stock', {
   608→        method: 'POST',
   609→        headers: { 'Content-Type': 'application/json' },
   610→        body: JSON.stringify({ ...values, type: 'out', userId: '' }),
   611→      });
   612→      const json = await res.json();
   613→      if (!res.ok) throw new Error(json.error || 'Gagal mengurangi stok');
   614→      return json;
   615→    },
   616→    onSuccess: () => {
   617→      toast.success('Stok keluar berhasil dicatat');
   618→      queryClient.invalidateQueries({ queryKey: ['stock-history'] });
   619→      queryClient.invalidateQueries({ queryKey: ['products-search'] });
   620→      queryClient.invalidateQueries({ queryKey: ['product', selectedProductId] });
   621→      form.reset();
   622→      setSelectedProductId('');
   623→    },
   624→    onError: (err) => toast.error(err.message),
   625→  });
   626→
   627→  const onSubmit = (values: StockOutForm) => mutation.mutate(values);
   628→
   629→  const currentStock = selectedProduct?.stock ?? 0;
   630→
   631→  return (

   633→      initial={{ opacity: 0, y: 10 }}
