    64→function formatRupiah(amount: number): string {
    65→  return new Intl.NumberFormat('id-ID', {
    66→    style: 'currency',
    67→    currency: 'IDR',
    68→    minimumFractionDigits: 0,
    69→    maximumFractionDigits: 0,
    70→  }).format(amount);
    71→}
    72→
    73→function formatDate(dateStr: string): string {
    74→  return new Date(dateStr).toLocaleDateString('id-ID', {
    75→    day: '2-digit',
    76→    month: 'short',
    77→    year: 'numeric',
    78→  });
    79→}

    81→function getDefaultDates() {
    82→  const now = new Date();
    83→  const firstDay = new Date(now.getFullYear(), now.getMonth(), 1);
    84→  return {
    85→    start: firstDay.toISOString().split('T')[0],
    86→    end: now.toISOString().split('T')[0],
    87→  };
    88→}

   539→function LabaRugiTab() {
   540→  const defaults = getDefaultDates();
   541→  const [startDate, setStartDate] = useState(defaults.start);
   542→  const [endDate, setEndDate] = useState(defaults.end);
   543→
   544→  const { data, isLoading, isError } = useQuery({
   545→    queryKey: ['report-profit-loss', startDate, endDate],
   546→    queryFn: async () => {
   547→      const params = new URLSearchParams({ type: 'profit_loss' });
   548→      if (startDate) params.set('startDate', startDate);
   549→      if (endDate) params.set('endDate', endDate);
   550→      const res = await fetch(`/api/reports?${params}`);
   551→      if (!res.ok) throw new Error('Gagal memuat data');
   552→      const json = await res.json();
   553→      return json.data;
   554→    },
   555→  });
   556→
   557→  // Also fetch daily data for chart
   558→  const { data: dailyData } = useQuery({
   559→    queryKey: ['report-daily-pl', startDate, endDate],
   560→    queryFn: async () => {
   561→      const params = new URLSearchParams({ type: 'daily' });
   562→      if (startDate) params.set('startDate', startDate);
   563→      if (endDate) params.set('endDate', endDate);
   564→      const res = await fetch(`/api/reports?${params}`);
   565→      if (!res.ok) return null;
   566→      const json = await res.json();
   567→      return json.data;
   568→    },
   569→  });
   570→
   571→  const chartDailyData = useMemo(() => {
   572→    if (!dailyData?.transactions) return [];
   573→    const grouped: Record<string, { date: string; laba: number; hpp: number }> = {};
   574→    for (const tx of dailyData.transactions) {
   575→      const day = new Date(tx.createdAt).toISOString().split('T')[0];
   576→      if (!grouped[day]) grouped[day] = { date: day, laba: 0, hpp: 0 };
   577→      const cost = tx.items?.reduce((s: number, item: { costPrice: number; quantity: number }) => s + item.costPrice * item.quantity, 0) ?? 0;
   578→      grouped[day].laba += tx.grandTotal - cost;

   999→          found.transaksi += 1;
  1000→        }
