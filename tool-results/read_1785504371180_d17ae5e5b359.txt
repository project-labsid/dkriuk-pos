    97→// ─── Form Schema ───────────────────────────────────────────────────────────────
    98→
    99→const customerFormSchema = z.object({
   100→  name: z.string().min(1, 'Nama pelanggan wajib diisi'),
   101→  phone: z.string().optional().or(z.literal('')),
   102→  email: z.string().email('Email tidak valid').optional().or(z.literal('')),
   103→  address: z.string().optional().or(z.literal('')),
   104→  memberPoint: z.coerce.number().int().min(0, 'Poin tidak boleh negatif').default(0),
   105→  isActive: z.boolean().default(true),
   106→});
   107→
   108→type CustomerFormData = z.infer<typeof customerFormSchema>;
   109→
   110→// ─── Animation Variants ──────────────────────────────────────────────────────
   111→
   112→const containerVariants = {
   113→  hidden: { opacity: 0 },
   114→  visible: {
   115→    opacity: 1,
   116→    transition: { staggerChildren: 0.08 },
   117→  },
   118→};
   119→
   120→const itemVariants = {
   121→  hidden: { opacity: 0, y: 20 },
   122→  visible: { opacity: 1, y: 0, transition: { duration: 0.4 } },
   123→};
   124→
   125→// ─── Status Badge ─────────────────────────────────────────────────────────────
   126→
   127→function getStatusBadge(isActive: boolean) {
   128→  if (isActive) {
   129→    return (
   130→      <Badge className="bg-emerald-100 text-emerald-700 border-emerald-200 hover:bg-emerald-100">
   131→        Aktif
   132→      </Badge>
   133→    );
   134→  }
   135→  return (
   136→    <Badge className="bg-gray-100 text-gray-600 border-gray-200 hover:bg-gray-100 dark:bg-gray-800 dark:text-gray-400 dark:border-gray-700">
   137→      Tidak Aktif
   138→    </Badge>
   139→  );
   140→}

   142→// ─── Main Component ────────────────────────────────────────────────────────────
   143→
   144→export default function CustomersPage() {
   145→  const queryClient = useQueryClient();
   146→
   147→  // ── Filter State ────────────────────────────────────────────────────────────
   148→  const [search, setSearch] = useState('');
   149→  const [page, setPage] = useState(1);
   150→  const [perPage, setPerPage] = useState(10);
   151→
   152→  // ── Dialog State ────────────────────────────────────────────────────────────
   153→  const [formOpen, setFormOpen] = useState(false);
   154→  const [deleteOpen, setDeleteOpen] = useState(false);
   155→  const [editingCustomer, setEditingCustomer] = useState<Customer | null>(null);
   156→  const [deletingCustomer, setDeletingCustomer] = useState<Customer | null>(null);
   157→
   158→  // ── Form ───────────────────────────────────────────────────────────────────
   159→  const form = useForm<CustomerFormData>({
   160→    resolver: zodResolver(customerFormSchema),
   161→    defaultValues: {
   162→      name: '',
   163→      phone: '',
   164→      email: '',
   165→      address: '',
   166→      memberPoint: 0,
   167→      isActive: true,
   168→    },
   169→  });
   170→
   171→  // ── Query ─────────────────────────────────────────────────────────────────
   172→
   173→  const customersQuery = useQuery<PaginatedResponse<Customer>>({
   174→    queryKey: ['customers', page, perPage, search],
   175→    queryFn: async () => {
   176→      const params = new URLSearchParams({
   177→        page: String(page),
   178→        perPage: String(perPage),
   179→      });
   180→      if (search) params.set('search', search);
   181→      const res = await fetch(`/api/customers?${params.toString()}`);
   182→      if (!res.ok) throw new Error('Gagal memuat pelanggan');
   183→      return res.json();
   184→    },
   185→  });
   186→
   187→  // ── Mutations ──────────────────────────────────────────────────────────────
   188→
   189→  const createMutation = useMutation({
   190→    mutationFn: async (data: CustomerFormData) => {
   191→      const payload: Record<string, unknown> = { ...data };
   192→      if (!payload.phone) delete payload.phone;
   193→      if (!payload.email) delete payload.email;
   194→      if (!payload.address) delete payload.address;
   195→      const res = await fetch('/api/customers', {
   196→        method: 'POST',
   197→        headers: { 'Content-Type': 'application/json' },
   198→        body: JSON.stringify(payload),
   199→      });
   200→      if (!res.ok) {
   201→        const err = await res.json();
   202→        throw new Error(err.error || 'Gagal menambah pelanggan');
   203→      }
   204→      return res.json();
   205→    },
   206→    onSuccess: () => {
   207→      toast.success('Pelanggan berhasil ditambahkan');
   208→      queryClient.invalidateQueries({ queryKey: ['customers'] });
   209→      closeFormDialog();
   210→    },
   211→    onError: (error) => {
   212→      toast.error(error.message);
   213→    },
   214→  });
   215→
   216→  const updateMutation = useMutation({
   217→    mutationFn: async (data: CustomerFormData) => {
   218→      if (!editingCustomer) return;
   219→      const payload: Record<string, unknown> = { ...data };
   220→      if (!payload.phone) payload.phone = null;
   221→      if (!payload.email) payload.email = null;
   222→      if (!payload.address) payload.address = null;
   223→      const res = await fetch(`/api/customers/${editingCustomer.id}`, {
   224→        method: 'PUT',
   225→        headers: { 'Content-Type': 'application/json' },
   226→        body: JSON.stringify(payload),
   227→      });
   228→      if (!res.ok) {
   229→        const err = await res.json();
   230→        throw new Error(err.error || 'Gagal memperbarui pelanggan');
   231→      }
   232→      return res.json();
   233→    },
   234→    onSuccess: () => {
   235→      toast.success('Pelanggan berhasil diperbarui');
   236→      queryClient.invalidateQueries({ queryKey: ['customers'] });
   237→      closeFormDialog();
   238→    },
   239→    onError: (error) => {
   240→      toast.error(error.message);
   241→    },
   242→  });
   243→
   244→  const deleteMutation = useMutation({
   245→    mutationFn: async (id: string) => {
   246→      const res = await fetch(`/api/customers/${id}`, {
   247→        method: 'DELETE',
   248→      });
   249→      if (!res.ok) {
   250→        const err = await res.json();
   251→        throw new Error(err.error || 'Gagal menghapus pelanggan');
   252→      }
   253→      return res.json();
   254→    },
   255→    onSuccess: () => {
   256→      toast.success('Pelanggan berhasil dihapus');
   257→      queryClient.invalidateQueries({ queryKey: ['customers'] });
   258→      setDeleteOpen(false);
   259→      setDeletingCustomer(null);
   260→    },
   261→    onError: (error) => {
   262→      toast.error(error.message);
   263→    },
   264→  });
   265→
   266→  // ── Derived Data ────────────────────────────────────────────────────────────
   267→
   268→  const customers = customersQuery.data?.data ?? [];
   269→  const totalCustomers = customersQuery.data?.total ?? 0;
   270→  const totalPages = customersQuery.data?.totalPages ?? 1;
   271→  const isLoading = customersQuery.isLoading || customersQuery.isFetching;
   272→  const isMutating = createMutation.isPending || updateMutation.isPending;
   273→
   274→  const startItem = (page - 1) * perPage + 1;
   275→  const endItem = Math.min(page * perPage, totalCustomers);
   276→
   277→  // ── Handlers ────────────────────────────────────────────────────────────────
   278→
   279→  const openAddDialog = useCallback(() => {
   280→    setEditingCustomer(null);
   281→    form.reset({
   282→      name: '',
   283→      phone: '',
   284→      email: '',
   285→      address: '',
   286→      memberPoint: 0,
   287→      isActive: true,
   288→    });
   289→    setFormOpen(true);
   290→  }, [form]);
   291→
   292→  const openEditDialog = useCallback(
   293→    (customer: Customer) => {
   294→      setEditingCustomer(customer);
   295→      form.reset({
   296→        name: customer.name,
   297→        phone: customer.phone ?? '',
   298→        email: customer.email ?? '',
   299→        address: customer.address ?? '',
   300→        memberPoint: customer.memberPoint,
   301→        isActive: customer.isActive,
   302→      });
   303→      setFormOpen(true);
   304→    },
   305→    [form]
   306→  );
   307→
   308→  const openDeleteDialog = useCallback((customer: Customer) => {
   309→    setDeletingCustomer(customer);
   310→    setDeleteOpen(true);
   311→  }, []);
   312→
   313→  const closeFormDialog = useCallback(() => {
   314→    setFormOpen(false);
   315→    setEditingCustomer(null);
   316→  }, []);
   317→
   318→  const onSubmitForm = useCallback(
   319→    (data: CustomerFormData) => {
   320→      if (editingCustomer) {
   321→        updateMutation.mutate(data);
   322→      } else {
   323→        createMutation.mutate(data);
   324→      }
   325→    },
   326→    [editingCustomer, createMutation, updateMutation]
   327→  );
   328→
   329→  const handleDeleteConfirm = useCallback(() => {
   330→    if (deletingCustomer) {
   331→      deleteMutation.mutate(deletingCustomer.id);
   332→    }
   333→  }, [deletingCustomer, deleteMutation]);
   334→
   335→  const handleSearchChange = useCallback((value: string) => {
   336→    setSearch(value);
   337→    setPage(1);
   338→  }, []);
   339→
   340→  const handlePerPageChange = useCallback((value: string) => {
   341→    setPerPage(Number(value));
   342→    setPage(1);
   343→  }, []);
   344→
   345→  // ── Table Columns ─────────────────────────────────────────────────────────
   346→
   347→  const columns = useMemo<ColumnDef<Customer>[]>(
   348→    () => [
   349→      {
   350→        accessorKey: 'index',
   351→        header: 'No',
   352→        size: 50,
   353→        cell: ({ row }) => (
   354→          <span className="text-muted-foreground text-sm">{row.index + 1}</span>
   355→        ),
   356→      },
   357→      {
   358→        accessorKey: 'name',
   359→        header: 'Nama',
   360→        cell: ({ row }) => (
   361→          <div className="font-medium">{row.original.name}</div>
   362→        ),
   363→      },
   364→      {
   365→        accessorKey: 'phone',
   366→        header: 'Telepon',
   367→        size: 130,
   368→        cell: ({ row }) => (
   369→          <span className="text-muted-foreground text-sm">
   370→            {row.original.phone || '-'}
   371→          </span>
   372→        ),
   373→      },
   374→      {
   375→        accessorKey: 'email',
   376→        header: 'Email',
   377→        size: 180,
   378→        cell: ({ row }) => (
   379→          <span className="text-muted-foreground text-sm truncate block max-w-[180px]">
   380→            {row.original.email || '-'}
   381→          </span>
   382→        ),
   383→      },
   384→      {
   385→        accessorKey: 'address',
   386→        header: 'Alamat',
   387→        size: 200,
   388→        cell: ({ row }) => (
   389→          <span className="text-muted-foreground text-sm truncate block max-w-[200px]">
   390→            {row.original.address || '-'}
   391→          </span>
   392→        ),
   393→      },
   394→      {
   395→        accessorKey: 'memberPoint',
   396→        header: 'Poin Member',
   397→        size: 120,
   398→        cell: ({ row }) => (
   399→          <Badge className="bg-amber-100 text-amber-700 border-amber-200 hover:bg-amber-100">
   400→            {row.original.memberPoint.toLocaleString('id-ID')} poin
   401→          </Badge>
   402→        ),
   403→      },
   404→      {
   405→        accessorKey: 'isActive',
   406→        header: 'Status',
   407→        size: 120,
   408→        cell: ({ row }) => getStatusBadge(row.original.isActive),
   409→      },
   410→      {
   411→        id: 'actions',
   412→        header: 'Aksi',
   413→        size: 80,
   414→        cell: ({ row }) => (
   415→          <DropdownMenu>
   416→            <DropdownMenuTrigger asChild>
   417→              <Button variant="ghost" size="icon" className="size-8">
   418→                <Pencil className="size-4" />
   419→                <span className="sr-only">Aksi</span>
   420→              </Button>
   421→            </DropdownMenuTrigger>
   422→            <DropdownMenuContent align="end">
   423→              <DropdownMenuItem onClick={() => openEditDialog(row.original)}>
   424→                <Pencil className="size-4 mr-2" />
   425→                Edit
   426→              </DropdownMenuItem>
   427→              <DropdownMenuSeparator />
   428→              <DropdownMenuItem
   429→                onClick={() => openDeleteDialog(row.original)}
   430→                className="text-destructive focus:text-destructive"
   431→              >
   432→                <Trash2 className="size-4 mr-2" />
   433→                Hapus
   434→              </DropdownMenuItem>
   435→            </DropdownMenuContent>
   436→          </DropdownMenu>
   437→        ),
   438→      },
   439→    ],
   440→    [openEditDialog, openDeleteDialog]
   441→  );
   442→
   443→  // ── Table ─────────────────────────────────────────────────────────────────
   444→
   445→  const table = useReactTable({
   446→    data: customers,
   447→    columns,
   448→    getCoreRowModel: getCoreRowModel(),
   449→    getPaginationRowModel: getPaginationRowModel(),
   450→    manualPagination: true,
   451→    pageCount: totalPages,
   452→    state: {
   453→      pagination: { pageIndex: page - 1, pageSize: perPage },
   454→    },
   455→  });
   456→
   457→  // ── Form Content ───────────────────────────────────────────────────────────
   458→
   459→  const formContent = (
   460→    <Form {...form}>
   461→      <form onSubmit={(e) => e.preventDefault()} className="space-y-4">
   462→        <FormField
   463→          control={form.control}
   464→          name="name"
   465→          render={({ field }) => (
   466→            <FormItem>
   467→              <FormLabel>Nama Pelanggan</FormLabel>
   468→              <FormControl>
   469→                <Input placeholder="Masukkan nama pelanggan" {...field} />
   470→              </FormControl>
   471→              <FormMessage />
   472→            </FormItem>
   473→          )}
   474→        />
   475→        <FormField
   476→          control={form.control}
   477→          name="phone"
   478→          render={({ field }) => (
   479→            <FormItem>
   480→              <FormLabel>Telepon</FormLabel>
   481→              <FormControl>
   482→                <Input placeholder="Nomor telepon pelanggan" {...field} />
   483→              </FormControl>
   484→              <FormMessage />
   485→            </FormItem>
   486→          )}
   487→        />
   488→        <FormField
   489→          control={form.control}
   490→          name="email"
   491→          render={({ field }) => (
   492→            <FormItem>
   493→              <FormLabel>Email</FormLabel>
   494→              <FormControl>
   495→                <Input
   496→                  type="email"
   497→                  placeholder="Email pelanggan (opsional)"
   498→                  {...field}
   499→                />
   500→              </FormControl>
   501→              <FormMessage />
   502→            </FormItem>
   503→          )}
   504→        />
   505→        <FormField
   506→          control={form.control}
   507→          name="address"
   508→          render={({ field }) => (
   509→            <FormItem>
   510→              <FormLabel>Alamat</FormLabel>
   511→              <FormControl>
   512→                <Textarea placeholder="Alamat lengkap pelanggan" {...field} />
   513→              </FormControl>
   514→              <FormMessage />
   515→            </FormItem>
   516→          )}
   517→        />
   518→        <FormField
   519→          control={form.control}
   520→          name="memberPoint"
   521→          render={({ field }) => (
   522→            <FormItem>
   523→              <FormLabel>Poin Member</FormLabel>
   524→              <FormControl>
   525→                <Input
   526→                  type="number"
   527→                  min={0}
   528→                  placeholder="0"
   529→                  {...field}
   530→                />
   531→              </FormControl>
   532→              <FormMessage />
   533→            </FormItem>
   534→          )}
   535→        />
   536→        <FormField
   537→          control={form.control}
   538→          name="isActive"
   539→          render={({ field }) => (
   540→            <FormItem className="flex items-center justify-between rounded-lg border p-3">
   541→              <div className="space-y-0.5">
   542→                <FormLabel>Status Aktif</FormLabel>
   543→                <p className="text-muted-foreground text-sm">
   544→                  Aktifkan pelanggan untuk digunakan
   545→                </p>
   546→              </div>
   547→              <FormControl>
   548→                <Switch
   549→                  checked={field.value}
   550→                  onCheckedChange={field.onChange}
   551→                />
   552→              </FormControl>
   553→            </FormItem>
   554→          )}
   555→        />
   556→      </form>
   557→    </Form>
   558→  );
   559→
   560→  // ── Render ─────────────────────────────────────────────────────────────────
   561→
   562→  return (
   563→    <motion.div
   564→      variants={containerVariants}
   565→      initial="hidden"
   566→      animate="visible"
   567→      className="flex flex-col gap-6 p-4 md:p-6"
   568→    >
   569→      {/* ── Header ─────────────────────────────────────────────────────────────── */}
   570→      <motion.div
   571→        variants={itemVariants}
   572→        className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4"
   573→      >
   574→        <div>
   575→          <h1 className="text-2xl font-bold tracking-tight">Pelanggan</h1>
   576→          <p className="text-muted-foreground text-sm mt-1">
   577→            Kelola data pelanggan toko Anda
   578→          </p>
   579→        </div>
   580→        <Button onClick={openAddDialog} className="bg-emerald-600 hover:bg-emerald-700">
   581→          <Plus className="size-4" />
   582→          Tambah Pelanggan
   583→        </Button>
   584→      </motion.div>
   585→
   586→      {/* ── Search & Table ────────────────────────────────────────────────────── */}
   587→      <motion.div
   588→        variants={itemVariants}
   589→        className="border rounded-xl bg-card shadow-sm"
   590→      >
   591→        <div className="p-4 border-b">
   592→          <div className="relative max-w-sm">
   593→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   594→            <Input
   595→              placeholder="Cari pelanggan..."
   596→              value={search}
   597→              onChange={(e) => handleSearchChange(e.target.value)}
   598→              className="pl-9"
   599→            />
   600→          </div>
   601→        </div>
   602→
   603→        <div className="overflow-x-auto">
   604→          <Table>
   605→            <TableHeader>
   606→              {table.getHeaderGroups().map((headerGroup) => (
   607→                <TableRow key={headerGroup.id}>
   608→                  {headerGroup.headers.map((header) => (
   609→                    <TableHead key={header.id} style={{ width: header.getSize() }}>
   610→                      {flexRender(header.column.columnDef.header, header.getContext())}
   611→                    </TableHead>
   612→                  ))}
   613→                </TableRow>
   614→              ))}
   615→            </TableHeader>
   616→            <TableBody>
   617→              {isLoading ? (
   618→                Array.from({ length: perPage }).map((_, i) => (
   619→                  <TableRow key={i}>
   620→                    <TableCell><Skeleton className="h-4 w-6" /></TableCell>
   621→                    <TableCell><Skeleton className="h-4 w-[150px]" /></TableCell>
   622→                    <TableCell><Skeleton className="h-4 w-[100px]" /></TableCell>
   623→                    <TableCell><Skeleton className="h-4 w-[140px]" /></TableCell>
   624→                    <TableCell><Skeleton className="h-4 w-[160px]" /></TableCell>
   625→                    <TableCell><Skeleton className="h-5 w-[90px] rounded-full" /></TableCell>
   626→                    <TableCell><Skeleton className="h-5 w-[80px] rounded-full" /></TableCell>
   627→                    <TableCell><Skeleton className="h-8 w-8 rounded-md" /></TableCell>
   628→                  </TableRow>
   629→                ))
   630→              ) : customers.length === 0 ? (
   631→                <TableRow>
   632→                  <TableCell colSpan={columns.length} className="h-40 text-center">
   633→                    <div className="flex flex-col items-center justify-center gap-2">
   634→                      <Users className="size-10 text-muted-foreground/50" />
   635→                      <p className="text-muted-foreground text-sm">
   636→                        {search
   637→                          ? 'Tidak ada pelanggan yang sesuai dengan pencarian'
   638→                          : 'Belum ada pelanggan. Klik "Tambah Pelanggan" untuk memulai.'}
   639→                      </p>
   640→                    </div>
   641→                  </TableCell>
   642→                </TableRow>
   643→              ) : (
   644→                table.getRowModel().rows.map((row) => (
   645→                  <TableRow key={row.id}>
   646→                    {row.getVisibleCells().map((cell) => (
   647→                      <TableCell key={cell.id}>
   648→                        {flexRender(cell.column.columnDef.cell, cell.getContext())}
   649→                      </TableCell>
   650→                    ))}
   651→                  </TableRow>
   652→                ))
   653→              )}
   654→            </TableBody>
   655→          </Table>
   656→        </div>
   657→
   658→        {/* ── Pagination ─────────────────────────────────────────────────────── */}
   659→        {totalCustomers > 0 && (
   660→          <div className="flex flex-col sm:flex-row items-center justify-between gap-3 px-4 py-3 border-t">
   661→            <div className="flex items-center gap-3 text-sm text-muted-foreground">
   662→              <span>
   663→                Menampilkan {startItem}-{endItem} dari {totalCustomers}
   664→              </span>
   665→              <Select value={String(perPage)} onValueChange={handlePerPageChange}>
   666→                <SelectTrigger className="w-[70px] h-8">
   667→                  <SelectValue />
   668→                </SelectTrigger>
   669→                <SelectContent>
   670→                  <SelectItem value="10">10</SelectItem>
   671→                  <SelectItem value="25">25</SelectItem>
   672→                  <SelectItem value="50">50</SelectItem>
   673→                </SelectContent>
   674→              </Select>
   675→              <span>per halaman</span>
   676→            </div>
   677→            <div className="flex items-center gap-2">
   678→              <Button
   679→                variant="outline"
   680→                size="sm"
   681→                onClick={() => setPage((p) => Math.max(1, p - 1))}
   682→                disabled={page <= 1}
   683→              >
   684→                <ChevronLeft className="size-4" />
   685→                <span className="sm:hidden">Sebelumnya</span>
   686→              </Button>
   687→              <div className="flex items-center gap-1 text-sm">
   688→                <span className="px-2 py-1 rounded-md bg-primary text-primary-foreground font-medium">
   689→                  {page}
   690→                </span>
   691→                <span className="text-muted-foreground">dari {totalPages}</span>
   692→              </div>
   693→              <Button
   694→                variant="outline"
   695→                size="sm"
   696→                onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   697→                disabled={page >= totalPages}
   698→              >
   699→                <span className="sm:hidden">Berikutnya</span>
   700→                <ChevronRight className="size-4" />
   701→              </Button>
   702→            </div>
   703→          </div>
   704→        )}
   705→      </motion.div>
   706→
   707→      {/* ── Add Dialog (Desktop) ──────────────────────────────────────────────── */}
   708→      <Dialog open={formOpen && !editingCustomer} onOpenChange={setFormOpen}>
   709→        <DialogContent className="sm:max-w-lg">
   710→          <DialogHeader>
   711→            <DialogTitle>Tambah Pelanggan Baru</DialogTitle>
   712→            <DialogDescription>
   713→              Isi detail pelanggan yang akan ditambahkan
   714→            </DialogDescription>
   715→          </DialogHeader>
   716→          {formContent}
   717→          <DialogFooter className="pt-4">
   718→            <Button variant="outline" onClick={closeFormDialog} disabled={isMutating}>
   719→              Batal
   720→            </Button>
   721→            <Button
   722→              type="button"
   723→              onClick={form.handleSubmit(onSubmitForm)}
   724→              disabled={isMutating}
   725→              className="bg-emerald-600 hover:bg-emerald-700"
   726→            >
   727→              {isMutating && <Loader2 className="size-4 animate-spin" />}
   728→              {isMutating ? 'Menyimpan...' : 'Simpan Pelanggan'}
   729→            </Button>
   730→          </DialogFooter>
   731→        </DialogContent>
   732→      </Dialog>
   733→
   734→      {/* ── Edit Dialog (Desktop) ────────────────────────────────────────────── */}
   735→      <Dialog open={formOpen && !!editingCustomer} onOpenChange={setFormOpen}>
   736→        <DialogContent className="sm:max-w-lg">
   737→          <DialogHeader>
   738→            <DialogTitle>Edit Pelanggan</DialogTitle>
   739→            <DialogDescription>
   740→              Ubah detail pelanggan {editingCustomer?.name}
   741→            </DialogDescription>
   742→          </DialogHeader>
   743→          {formContent}
   744→          <DialogFooter className="pt-4">
   745→            <Button variant="outline" onClick={closeFormDialog} disabled={isMutating}>
   746→              Batal
   747→            </Button>
   748→            <Button
   749→              type="button"
   750→              onClick={form.handleSubmit(onSubmitForm)}
   751→              disabled={isMutating}
   752→              className="bg-emerald-600 hover:bg-emerald-700"
   753→            >
   754→              {isMutating && <Loader2 className="size-4 animate-spin" />}
   755→              {isMutating ? 'Menyimpan...' : 'Perbarui Pelanggan'}
   756→            </Button>
   757→          </DialogFooter>
   758→        </DialogContent>
   759→      </Dialog>
   760→
   761→      {/* ── Add/Edit Sheet (Mobile) ──────────────────────────────────────────── */}
   762→      <Sheet open={formOpen} onOpenChange={setFormOpen}>
   763→        <SheetContent side="bottom" className="sm:max-h-[85vh] rounded-t-2xl">
   764→          <SheetHeader>
   765→            <SheetTitle>
   766→              {editingCustomer ? 'Edit Pelanggan' : 'Tambah Pelanggan Baru'}
   767→            </SheetTitle>
   768→            <SheetDescription>
   769→              {editingCustomer
   770→                ? 'Ubah detail pelanggan'
   771→                : 'Isi detail pelanggan yang akan ditambahkan'}
   772→            </SheetDescription>
   773→          </SheetHeader>
   774→          <div className="flex-1 overflow-y-auto px-4 pb-4">
   775→            {formContent}
   776→          </div>
   777→          <SheetFooter className="flex-row gap-2 border-t pt-4">
   778→            <Button
   779→              variant="outline"
   780→              className="flex-1"
   781→              onClick={closeFormDialog}
   782→              disabled={isMutating}
   783→            >
   784→              Batal
   785→            </Button>
   786→            <Button
   787→              type="button"
   788→              className="flex-1 bg-emerald-600 hover:bg-emerald-700"
   789→              onClick={form.handleSubmit(onSubmitForm)}
   790→              disabled={isMutating}
   791→            >
   792→              {isMutating && <Loader2 className="size-4 animate-spin" />}
   793→              {isMutating ? 'Menyimpan...' : 'Simpan'}
   794→            </Button>
   795→          </SheetFooter>
   796→        </SheetContent>
   797→      </Sheet>
   798→
   799→      {/* ── Delete Confirmation Dialog ────────────────────────────────────────── */}
   800→      <AlertDialog open={deleteOpen} onOpenChange={setDeleteOpen}>
   801→        <AlertDialogContent>
   802→          <AlertDialogHeader>
   803→            <AlertDialogTitle>Hapus Pelanggan?</AlertDialogTitle>
   804→            <AlertDialogDescription>
   805→              Pelanggan <span className="font-semibold">{deletingCustomer?.name}</span>{' '}
   806→              akan dihapus secara permanen. Tindakan ini tidak dapat dibatalkan.
   807→            </AlertDialogDescription>
   808→          </AlertDialogHeader>
   809→          <AlertDialogFooter>
   810→            <AlertDialogCancel disabled={deleteMutation.isPending}>
   811→              Batal
   812→            </AlertDialogCancel>
   813→            <AlertDialogAction
   814→              onClick={handleDeleteConfirm}
   815→              disabled={deleteMutation.isPending}
   816→              className="bg-destructive hover:bg-destructive/90"
   817→            >
   818→              {deleteMutation.isPending && <Loader2 className="size-4 animate-spin" />}
   819→              {deleteMutation.isPending ? 'Menghapus...' : 'Hapus'}
   820→            </AlertDialogAction>
   821→          </AlertDialogFooter>
   822→        </AlertDialogContent>
   823→      </AlertDialog>
   824→    </motion.div>
   825→  );
   826→}
