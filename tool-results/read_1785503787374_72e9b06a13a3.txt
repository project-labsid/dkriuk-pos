   135→   135→function getInitials(name: string): string {
   136→   136→  return name
   137→   137→    .split(' ')
   138→   138→    .map((w) => w.charAt(0))
   139→   139→    .join('')
   140→   140→    .toUpperCase()
   141→   141→    .slice(0, 2);
   142→   142→}

   155→   155→function getColorForName(name: string): string {
   156→   156→  let hash = 0;
   157→   157→  for (let i = 0; i < name.length; i++) {
   158→   158→    hash = name.charCodeAt(i) + ((hash << 5) - hash);
   159→   159→  }
   160→   160→  return COLOR_PALETTE[Math.abs(hash) % COLOR_PALETTE.length];
   161→   161→}
