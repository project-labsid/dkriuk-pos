    65→function formatRupiah(amount: number): string {
    66→  return new Intl.NumberFormat('id-ID', {
    67→    style: 'currency',
    68→    currency: 'IDR',
    69→    minimumFractionDigits: 0,
    70→    maximumFractionDigits: 0,
    71→  }).format(amount);
    72→}
    73→
    74→function formatDate(dateStr: string): string {
    75→  return new Date(dateStr).toLocaleDateString('id-ID', {
    76→    day: '2-digit',
    77→    month: 'short',
    78→    year: 'numeric',
    79→    hour: '2-digit',
    80→    minute: '2-digit',
    81→  });
    82→}
    83→
    84→function formatDateShort(dateStr: string): string {
    85→  return new Date(dateStr).toLocaleDateString('id-ID', {
    86→    day: '2-digit',
    87→    month: 'short',
    88→    year: 'numeric',
    89→  });
    90→}

   107→// ─── Status / Payment Badges ─────────────────────────────────────────────────

   109→function getStatusBadge(status: string) {
   110→  switch (status) {
   111→    case 'completed':
   112→      return (
   113→        <Badge className="bg-emerald-100 text-emerald-700 border-emerald-200 hover:bg-emerald-100">
   114→          Selesai
   115→        </Badge>
   116→      );
   117→    case 'cancelled':
   118→      return (
   119→        <Badge className="bg-red-100 text-red-700 border-red-200 hover:bg-red-100">
   120→          Dibatalkan
   121→        </Badge>
   122→      );
   123→    case 'held':
   124→      return (
   125→        <Badge className="bg-amber-100 text-amber-700 border-amber-200 hover:bg-amber-100">
   126→          Ditahan
   127→        </Badge>
   128→      );
   129→    default:
   130→      return <Badge variant="secondary">{status}</Badge>;
   131→  }
   132→}

   619→                  type="date"
   620→                  value={endDate}
   621→                  onChange={(e) => handleEndDateChange(e.target.value)}
   622→                  className="pl-9"
   623→                />
   624→              </div>
   625→            </div>
   626→            {hasActiveFilters && (

   632→          </div>

   635→
   636→      {/* ── Search & Table ──────────────────────────────────────────────────── */}

   650→          </div>
   651→        </div>
   652→
   653→        <div className="overflow-x-auto">

   663→                </TableRow>

   665→            </TableHeader>
   666→            <TableBody>

   670→                    <TableCell><Skeleton className="h-4 w-6" /></TableCell>
   671→                    <TableCell><Skeleton className="h-4 w-[130px]" /></TableCell>
   672→                    <TableCell><Skeleton className="h-4 w-[140px]" /></TableCell>
   673→                    <TableCell><Skeleton className="h-4 w-[100px]" /></TableCell>

   683→                  <TableCell colSpan={columns.length} className="h-40 text-center">
   684→                    <div className="flex flex-col items-center justify-center gap-2">

   686→                      <p className="text-muted-foreground text-sm">

   689→                          : 'Belum ada riwayat transaksi.'}
   690→                      </p>
   691→                    </div>
   692→                  </TableCell>
   693→                </TableRow>
   694→              ) : (
   695→                table.getRowModel().rows.map((row) => (
   696→                  <TableRow
   697→                    key={row.id}
   698→                    className="cursor-pointer hover:bg-muted/50"
   699→                    onClick={() => openDetail(row.original.id)}
   700→                  >

   702→                      <TableCell key={cell.id}>
   703→                        {flexRender(cell.column.columnDef.cell, cell.getContext())}
   704→                      </TableCell>
   705→                    ))}
   706→                  </TableRow>
   707→                ))
   708→              )}
   709→            </TableBody>
   710→          </Table>
   711→        </div>
   712→
   713→        {/* ── Pagination ─────────────────────────────────────────────────── */}
   714→        {totalTransactions > 0 && (
   715→          <div className="flex flex-col sm:flex-row items-center justify-between gap-3 px-4 py-3 border-t">
   716→            <div className="flex items-center gap-3 text-sm text-muted-foreground">
   717→              <span>
   718→                Menampilkan {startItem}-{endItem} dari {totalTransactions}
   719→              </span>
   720→              <Select value={String(perPage)} onValueChange={handlePerPageChange}>
   721→                <SelectTrigger className="w-[70px] h-8">
   722→                  <SelectValue />
   723→                </SelectTrigger>
   724→                <SelectContent>
   725→                  <SelectItem value="10">10</SelectItem>
   726→                  <SelectItem value="25">25</SelectItem>
   727→                  <SelectItem value="50">50</SelectItem>
   728→                </SelectContent>
   729→              </Select>
   730→              <span>per halaman</span>
   731→            </div>
   732→            <div className="flex items-center gap-2">
   733→              <Button
   734→                variant="outline"
   735→                size="sm"
   736→                onClick={() => setPage((p) => Math.max(1, p - 1))}
   737→                disabled={page <= 1}
   738→              >
   739→                <ChevronLeft className="size-4" />
   740→                <span className="sm:hidden">Sebelumnya</span>
   741→              </Button>
   742→              <div className="flex items-center gap-1 text-sm">
   743→                <span className="px-2 py-1 rounded-md bg-primary text-primary-foreground font-medium">
   744→                  {page}
   745→                </span>
   746→                <span className="text-muted-foreground">dari {totalPages}</span>
   747→              </div>
   748→              <Button
   749→                variant="outline"
   750→                size="sm"
   751→                onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   752→                disabled={page >= totalPages}
   753→              >
   754→                <span className="sm:hidden">Berikutnya</span>
   755→                <ChevronRight className="size-4" />
   756→              </Button>
   757→            </div>
   758→          </div>
   759→        )}
   760→      </motion.div>
   761→
   762→      {/* ── Detail Dialog ──────────────────────────────────────────────────── */}
   763→      <Dialog open={detailOpen} onOpenChange={closeDetail}>
   764→        <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto">
   765→          <DialogHeader>
   766→            <DialogTitle>Detail Transaksi</DialogTitle>
   767→            <DialogDescription>
   768→              Rincian lengkap transaksi {detailData?.invoiceNumber}
   769→            </DialogDescription>
   770→          </DialogHeader>
   771→          {detailContent}
   772→        </DialogContent>
   773→      </Dialog>
   774→    </motion.div>
   775→  );
